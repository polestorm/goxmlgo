const (
	KindStartElement Kind = iota + 1
	KindEndElement
	KindText
	KindComment
	KindProcInst
	KindDoctype
)

type Attr struct {
	Name  Name
	Value string

	Type string

	Specified    bool
	Line, Column int
}

type Binding struct{ Prefix, URI string }

type Declaration struct {
	Version string

	Encoding string

	Standalone string

	InputEncoding string
	BOM           bool
}

type Error struct {
	Code string

	Line, Column int
	Message      string
}

type Kind uint8

type Limits struct {
	MaxDepth           int
	MaxNameLength      int
	MaxAttributes      int
	MaxTextSize        int
	MaxEntityExpansion int
	MaxEntityDepth     int
	MaxDTDSize         int

	MaxElements int

	MaxNamespaces int
}

type Name struct {
	Space, Local, Prefix string
}

type NamespaceScope struct {
	// contains filtered or unexported fields
}

type Notation struct {
	Name, PublicID, SystemID string
	HasPublic, HasSystem     bool
}

type Options struct {
	Limits Limits

	NoNamespaces bool
}

type Reader struct {
	// contains filtered or unexported fields
}

type Token struct {
	Kind  Kind
	Name  Name
	Attrs []Attr

	Namespaces []Binding

	Data []byte

	CDATA, Whitespace bool

	SelfClosing bool

	Line, Column int

	EndLine, EndColumn int
}

type TokenReader interface {
	Next() (Token, error)
}

func DefaultLimits() Limits

func New(r io.Reader, opts Options) *Reader

func (e *Error) Error() string

func (s *NamespaceScope) Depth() int

func (s *NamespaceScope) Lookup(prefix string) (string, bool)

func (s *NamespaceScope) Pop()

func (s *NamespaceScope) Push(declarations []Binding)

func (s *NamespaceScope) Reset()

func (s *NamespaceScope) Update(t Token)

func (r *Reader) Declaration() (Declaration, error)

func (r *Reader) DeclaredNamespaces() []Binding

func (r *Reader) InScopeNamespaces() []Binding

func (r *Reader) LookupNamespace(prefix string) (string, bool)

func (r *Reader) Next() (Token, error)

func (r *Reader) Notations() []Notation

func (r *Reader) Reset(src io.Reader, opts Options)

func (r *Reader) Skip() error

func (r *Reader) UnparsedEntity(name string) bool
